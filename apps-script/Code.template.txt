const DATABASE_ID = '1f_qveJu-Q0b9PiwrV1JDGCPeTytZOwhso0KHNTxzOoY';
const STORE_OPTIONS = __STORES__;
const WISH_TYPES = ['想合作的店家','想要的優惠','網站建議'];
const HEADERS = ['id','createdAt','status','storeId','storeName','nickname','rating','type','content','requestId'];
function setupCommunity(){const ss=SpreadsheetApp.openById(DATABASE_ID);['Reviews','Wishes'].forEach(name=>{let s=ss.getSheetByName(name);if(!s)s=ss.insertSheet(name);if(!s.getLastRow()){s.appendRow(HEADERS);s.setFrozenRows(1);s.getRange(1,1,1,HEADERS.length).setBackground('#BA3E25').setFontColor('#FFFFFF').setFontWeight('bold');s.setColumnWidth(9,420);s.getRange('C2:C10000').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['pending','approved','hidden'],true).setAllowInvalid(false).build());}else if(JSON.stringify(s.getRange(1,1,1,HEADERS.length).getValues()[0])!==JSON.stringify(HEADERS))throw Error(name+' 欄位不符，請勿覆蓋既有資料');});return '資料庫已就緒';}
function doGet(e){const id=String(e&&e.parameter&&e.parameter.storeId||'');return HtmlService.createHtmlOutput(PAGE_HTML.replace('__INITIAL_STORE__',JSON.stringify(STORE_OPTIONS.some(s=>s.id===id)?id:''))).setTitle('臺科特約留言與許願').setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);}
function readRows_(name){const s=SpreadsheetApp.openById(DATABASE_ID).getSheetByName(name);if(!s)throw Error('資料庫尚未初始化');const last=s.getLastRow();if(last<2)return [];const start=Math.max(2,last-1999);return s.getRange(start,1,last-start+1,HEADERS.length).getValues();}
function getCommunity(){return {reviews:readRows_('Reviews').filter(r=>r[2]==='approved').slice(-200).reverse().map(publicRow_),wishes:readRows_('Wishes').filter(r=>r[2]==='approved').slice(-200).reverse().map(publicRow_)};}
function publicRow_(r){return {id:String(r[0]),date:r[1] instanceof Date?r[1].toISOString():String(r[1]),storeId:String(r[3]),storeName:String(r[4]),nickname:String(r[5]),rating:Number(r[6])||0,type:String(r[7]),content:String(r[8])};}
function clean_(v,max){if(typeof v!=='string')throw Error('資料格式不正確');const s=v.trim();if(!s||s.length>max)throw Error('請確認必填欄位與字數');return s;}
function literal_(s){return /^[=+\-@\t\r]/.test(s)?"'"+s:s;}
function submitCommunity(p){if(!p||p.website)throw Error('無法送出');if(!['review','wish'].includes(p.kind))throw Error('類型不正確');const nickname=clean_(p.nickname,30),content=clean_(p.content,1000),requestId=clean_(p.requestId,80),client=clean_(p.clientId,80);if(!/^[a-zA-Z0-9-]{16,80}$/.test(requestId)||!/^[a-zA-Z0-9-]{16,80}$/.test(client))throw Error('請重新整理後再試');const store=STORE_OPTIONS.find(s=>s.id===p.storeId);let rating=0,type='';if(p.kind==='review'){rating=Number(p.rating);if(!store||!Number.isInteger(rating)||rating<1||rating>5)throw Error('請選擇店家及1至5顆星');}else{type=clean_(p.type,30);if(!WISH_TYPES.includes(type))throw Error('請選擇許願分類');}const lock=LockService.getScriptLock();lock.waitLock(10000);try{const name=p.kind==='review'?'Reviews':'Wishes';if(readRows_(name).some(r=>r[9]===requestId))return {ok:true,message:'這筆內容已送出，等待審核。'};const cache=CacheService.getScriptCache(),key='submit:'+client;if(cache.get(key))throw Error('送出太頻繁，請稍候一分鐘再試');const sheet=SpreadsheetApp.openById(DATABASE_ID).getSheetByName(name);sheet.appendRow([Utilities.getUuid(),new Date(),'pending',store?store.id:'',store?store.name:'',literal_(nickname),rating,type,literal_(content),requestId]);cache.put(key,'1',60);return {ok:true,message:'已送出！學生會審核後會公開顯示。'};}finally{lock.releaseLock();}}
const PAGE_HTML = __HTML__;
